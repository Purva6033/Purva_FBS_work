	create procedure Create_Account(
	in p_account_no int,
	in p_customer_name varchar(30),
	in p_opening_balance decimal(10,2),
	in p_account_type varchar(20))
	
begin
	declare cnt int;
	select count(*)
	into cnt
	from bank_account
	where account_no = p_account_no ;

	if cnt > 0  then
	rollback;
	select 'Account already exists' as Message;

	elseif p_opening_balance < 0 then
	rollback;
	select 'Opening balance cannott be negative ' as message;
	
	elseif p_account_type not in ('Saving' , 'Current') then
	rollback;
	select 'Invalid account type ' as message ;

	else 
	insert into bank_account 
	values (
	p_account_no,
	p_customer_name;
	p_opening_balance;
	p_account_type );

	commit;
	select 'account created successfully' as message;
	end if;
   	end //

	delimiter ;
